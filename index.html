""" News-Driven Stock Trading Bot v2 ================================== Improvements over v1: - Diversified ticker selection (no single stock dominance) - 3% stop-loss on all positions (industry standard for small accounts) - Per-ticker cooldown (same stock can't trade twice in a row) - Weighted sentiment scoring (stronger headlines = bigger signal) - Scans ALL headlines, scores ALL tickers, picks the BEST signals - Sends SMS alerts via Gmail -> Verizon vtext.com - Auto-pauses at $1,000 profit """ import time import smtplib import logging import requests from email.mime.text import MIMEText from datetime import datetime from alpaca.trading.client import TradingClient from alpaca.trading.requests import MarketOrderRequest from alpaca.trading.enums import OrderSide, TimeInForce from alpaca.data.historical import StockHistoricalDataClient from alpaca.data.requests import StockLatestQuoteRequest # ───────────────────────────────────────────── # CONFIGURATION -- Fill these in privately # ───────────────────────────────────────────── ALPACA_API_KEY = "YOUR_ALPACA_API_KEY" ALPACA_SECRET_KEY = "YOUR_ALPACA_SECRET_KEY" ALPACA_PAPER = True # Set False when going live NEWS_API_KEY = "YOUR_NEWSAPI_KEY" GMAIL_ADDRESS = "YOUR_GMAIL@gmail.com" GMAIL_APP_PASSWORD = "YOUR_16_CHAR_APP_PASSWORD" PHONE_NUMBER = "YOUR10DIGITNUMBER" VTEXT_ADDRESS = f"{PHONE_NUMBER}@vtext.com" STARTING_BALANCE = 50.0 MILESTONE_EVERY = 100.0 WITHDRAW_AT = 1000.0 STOP_LOSS_PCT = 0.03 # 3% stop-loss MAX_POSITIONS = 4 # Max stocks held at once TRADE_SIZE_PCT = 0.20 # 20% of cash per trade MAX_TRADE_USD = 12.0 # Hard cap per trade COOLDOWN_CYCLES = 2 # Cycles before same ticker can trade again # ───────────────────────────────────────────── # Logging # ───────────────────────────────────────────── logging.basicConfig( level=logging.INFO, format="%(asctime)s [%(levelname)s] %(message)s", handlers=[ logging.FileHandler("bot.log"), logging.StreamHandler() ] ) log = logging.getLogger(__name__) # ───────────────────────────────────────────── # Clients # ───────────────────────────────────────────── trading_client = TradingClient(ALPACA_API_KEY, ALPACA_SECRET_KEY, paper=ALPACA_PAPER) data_client = StockHistoricalDataClient(ALPACA_API_KEY, ALPACA_SECRET_KEY) # ───────────────────────────────────────────── # Sentiment word lists with weights # ───────────────────────────────────────────── POSITIVE_WORDS = { "surge": 2, "soar": 2, "record": 2, "beat": 2, "blowout": 3, "profit": 1, "growth": 1, "gain": 1, "rally": 2, "boost": 1, "bullish": 2, "upgrade": 2, "buy": 1, "strong": 1, "rise": 1, "breakthrough": 2, "expansion": 1, "partnership": 1, "deal": 1, "outperform": 2, "exceed": 2, "innovation": 1, "demand": 1, } NEGATIVE_WORDS = { "crash": 3, "fall": 1, "loss": 2, "miss": 2, "drop": 1, "recession": 3, "bearish": 2, "downgrade": 2, "sell": 1, "weak": 1, "decline": 1, "layoff": 2, "bankrupt": 3, "risk": 1, "concern": 1, "warning": 2, "lawsuit": 2, "investigation": 2, "recall": 2, "tariff": 1, "fine": 1, "penalty": 2, "hack": 2, "breach": 2, } # ───────────────────────────────────────────── # Ticker keyword map # ───────────────────────────────────────────── TICKER_KEYWORDS = { "apple": "AAPL", "iphone": "AAPL", "microsoft": "MSFT", "azure": "MSFT", "amazon": "AMZN", "aws": "AMZN", "google": "GOOGL", "alphabet": "GOOGL", "nvidia": "NVDA", "gpu": "NVDA", "meta": "META", "facebook": "META", "instagram": "META", "tesla": "TSLA", "elon": "TSLA", "netflix": "NFLX", "salesforce": "CRM", "jpmorgan": "JPM", "bank": "JPM", "goldman": "GS", "fed": "SPY", "federal reserve": "SPY", "interest rate": "SPY", "oil": "XOM", "exxon": "XOM", "energy": "XOM", "opec": "XOM", "pfizer": "PFE", "vaccine": "PFE", "drug": "JNJ", "fda": "JNJ", "economy": "SPY", "market": "SPY", "nasdaq": "QQQ", "tech": "QQQ", "recession": "SPY", "tariff": "SPY", "trade war": "SPY", } # ───────────────────────────────────────────── # SMS # ───────────────────────────────────────────── def send_sms(message: str): try: msg = MIMEText(message) msg["From"] = GMAIL_ADDRESS msg["To"] = VTEXT_ADDRESS msg["Subject"] = "" with smtplib.SMTP_SSL("smtp.gmail.com", 465) as server: server.login(GMAIL_ADDRESS, GMAIL_APP_PASSWORD) server.sendmail(GMAIL_ADDRESS, VTEXT_ADDRESS, msg.as_string()) log.info(f"SMS sent: {message}") except Exception as e: log.error(f"SMS failed: {e}") # ───────────────────────────────────────────── # News signal scoring # ───────────────────────────────────────────── def get_news_signals(): """ Score ALL tickers across ALL headlines. Returns list sorted by signal strength (strongest first). Caps each ticker's per-headline contribution to prevent keyword-heavy tickers (like Apple) from dominating. """ url = ( f"https://newsapi.org/v2/top-headlines" f"?category=business&language=en&pageSize=30&apiKey={NEWS_API_KEY}" ) try: resp = requests.get(url, timeout=10) articles = resp.json().get("articles", []) except Exception as e: log.error(f"News fetch failed: {e}") return [] ticker_scores = {} ticker_reasons = {} for article in articles: title = (article.get("title") or "").lower() desc = (article.get("description") or "").lower() text = title + " " + desc matched = {} for keyword, sym in TICKER_KEYWORDS.items(): if keyword in text: matched[sym] = matched.get(sym, 0) + 1 if not matched: continue score = 0 for word, weight in POSITIVE_WORDS.items(): if word in text: score += weight for word, weight in NEGATIVE_WORDS.items(): if word in text: score -= weight if score == 0: continue for sym in matched: # Cap contribution per headline at +/-3 contribution = min(abs(score), 3) * (1 if score > 0 else -1) ticker_scores[sym] = ticker_scores.get(sym, 0) + contribution if sym not in ticker_reasons: ticker_reasons[sym] = article.get("title", "") signals = [] for sym, total_score in ticker_scores.items(): if total_score != 0: sentiment = "BUY" if total_score > 0 else "SELL" signals.append((sym, sentiment, total_score, ticker_reasons.get(sym, ""))) log.info(f"Signal: {sentiment:4} {sym:6} score={total_score:+d}") signals.sort(key=lambda x: abs(x[2]), reverse=True) return signals # ───────────────────────────────────────────── # Alpaca helpers # ───────────────────────────────────────────── def get_account(): return trading_client.get_account() def get_portfolio_value(): return float(get_account().portfolio_value) def get_current_price(ticker: str) -> float: req = StockLatestQuoteRequest(symbol_or_symbols=ticker) quote = data_client.get_stock_latest_quote(req) return float(quote[ticker].ask_price) def get_all_positions() -> dict: try: return {p.symbol: p for p in trading_client.get_all_positions()} except Exception as e: log.error(f"Error fetching positions: {e}") return {} def place_order(ticker: str, side: OrderSide, dollars: float): try: price = get_current_price(ticker) qty = round(dollars / price, 6) if qty < 0.001: log.info(f"Order too small for {ticker}, skipping.") return None req = MarketOrderRequest( symbol=ticker, qty=qty, side=side, time_in_force=TimeInForce.DAY, ) order = trading_client.submit_order(req) log.info(f"Order placed: {side.value.upper()} {qty:.4f} {ticker} @ ~${price:.2f}") return order except Exception as e: log.error(f"Order failed {ticker}: {e}") return None def close_position(ticker: str): try: trading_client.close_position(ticker) log.info(f"Closed position: {ticker}") except Exception as e: log.error(f"Could not close {ticker}: {e}") # ───────────────────────────────────────────── # Stop-loss checker — runs every cycle # ───────────────────────────────────────────── def check_stop_losses(positions: dict) -> list: """ Close any position down more than STOP_LOSS_PCT from entry. Standard 3% stop-loss protects capital on small accounts. """ closed = [] for sym, pos in positions.items(): try: avg_entry = float(pos.avg_entry_price) curr_price = float(pos.current_price) loss_pct = (curr_price - avg_entry) / avg_entry if loss_pct <= -STOP_LOSS_PCT: log.warning( f"STOP-LOSS: {sym} entry=${avg_entry:.2f} " f"now=${curr_price:.2f} ({loss_pct*100:.1f}%)" ) close_position(sym) closed.append(sym) send_sms( f"Stop-loss hit: {sym} " f"({loss_pct*100:.1f}% loss). Position closed." ) except Exception as e: log.error(f"Stop-loss check error {sym}: {e}") return closed # ───────────────────────────────────────────── # Main bot loop # ───────────────────────────────────────────── def run_bot(starting_balance: float): log.info(f"Bot v2 started | Balance: ${starting_balance:.2f} | Stop-loss: {STOP_LOSS_PCT*100:.0f}%") send_sms(f"Bot started with ${starting_balance:.2f}. 3% stop-loss active. Diversified trading enabled.") last_milestone = 0.0 cooldown_map = {} # ticker -> cycles remaining on cooldown cycle = 0 while True: try: cycle += 1 log.info(f"\n--- Cycle {cycle} | {datetime.now().strftime('%H:%M:%S')} ---") # Tick down cooldowns cooldown_map = {k: v-1 for k, v in cooldown_map.items() if v > 1} # Portfolio check portfolio_value = get_portfolio_value() profit = portfolio_value - starting_balance log.info(f"Portfolio: ${portfolio_value:.2f} | Profit: ${profit:+.2f}") # Withdraw threshold if profit >= WITHDRAW_AT: send_sms( f"Bot hit ${WITHDRAW_AT:.0f} profit! " f"Portfolio: ${portfolio_value:.2f}. " f"Paused. Reply with new amount to restart." ) for sym in get_all_positions(): close_position(sym) log.info("Withdraw threshold hit. Bot stopped.") break # Milestone alerts if profit > 0: milestone_hit = int(profit / MILESTONE_EVERY) * MILESTONE_EVERY if milestone_hit > last_milestone: send_sms( f"+${milestone_hit:.0f} profit milestone! " f"Portfolio: ${portfolio_value:.2f}" ) last_milestone = milestone_hit # Stop-loss check positions = get_all_positions() closed = check_stop_losses(positions) for sym in closed: cooldown_map[sym] = COOLDOWN_CYCLES # Refresh after stop-losses positions = get_all_positions() # Get news signals signals = get_news_signals() if not signals: log.info("No actionable signals this cycle.") else: account = get_account() buying_power = float(account.buying_power) trade_size = min(buying_power * TRADE_SIZE_PCT, MAX_TRADE_USD) buys_made = 0 max_new_buys = MAX_POSITIONS - len(positions) for ticker, sentiment, score, headline in signals: # Skip cooldown tickers if ticker in cooldown_map: log.info(f"Cooldown: {ticker} ({cooldown_map[ticker]} cycles left)") continue if sentiment == "BUY": if buys_made >= max_new_buys: continue if ticker in positions: log.info(f"Already holding {ticker}") continue if trade_size >= 1.0: result = place_order(ticker, OrderSide.BUY, trade_size) if result: buys_made += 1 cooldown_map[ticker] = COOLDOWN_CYCLES log.info(f"Buy reason: {headline[:70]}") elif sentiment == "SELL": if ticker in positions: close_position(ticker) cooldown_map[ticker] = COOLDOWN_CYCLES log.info(f"Sell reason: {headline[:70]}") log.info(f"Sleeping 15 min | Active cooldowns: {list(cooldown_map.keys())}") time.sleep(15 * 60) except KeyboardInterrupt: log.info("Bot manually stopped.") send_sms("Bot manually stopped.") break except Exception as e: log.error(f"Unexpected error: {e}") time.sleep(60) # ───────────────────────────────────────────── # Entry point # ───────────────────────────────────────────── if __name__ == "__main__": run_bot(STARTING_BALANCE)
